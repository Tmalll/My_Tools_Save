@echo off
chcp 65001 >nul
echo 开始执行 [ %~nx0 ]
title [ %~nx0 ]
echo.

:: 最小化窗口
cd /d "%~dp0" & title %~nx0
if "%1" neq "min" start /min "" "%~f0" min & exit
:start_script
:: 这里放后续脚本...

:: 日志文件路径
set "LOGpath=E:\01.userData\ZhuoMian\"

:: 排除当前脚本, 并且顺序运行当前目录下面其他.bat脚本
for %%f in (*.bat) do (
    if /I not "%%~nxf"=="%~nx0" (
        echo.
        echo = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = 
        echo.
        echo 正在运行: %%f
        echo.
        echo = = = = = = = = = = = = = = = = = = = = = = = = = = = = = = 
        echo.
        timeout /t 1 > NUL
        start /min "" cmd /c ""%%f" > "%LOGpath%%%~nf.log" 2>&1"
        echo.
    )
)


echo 所有 [ Rclone 同步脚本 ] 已执行完毕, 10秒后退出脚本.
echo.
timeout /t 10
:: pause
exit
