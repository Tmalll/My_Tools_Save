# 查询是否已存在该 DNS 记录（A/AAAA/cname等记录）
RECORD_ID=$(curl -s -X GET "https://api.cloudflare.com/client/v4/zones/$ZONE_ID/dns_records?type=AAAA&name=$HOSTNAME" \
     -H "Authorization: Bearer $CF_API_TOKEN" \
     -H "Content-Type: application/json" | jq -r '.result[0].id')
这里在增强一下,  如果ip地址一样,  就直接跳过此次DDNS脚本...

# 如果记录已存在，更新该记录
if [ "$RECORD_ID" != "null" ]; then
  echo "Found existing DNS record. Updating IPv6 address..."
  curl -s -X PUT "https://api.cloudflare.com/client/v4/zones/$ZONE_ID/dns_records/$RECORD_ID" \
       -H "Authorization: Bearer $CF_API_TOKEN" \
       -H "Content-Type: application/json" \
       --data "{\"type\":\"AAAA\",\"name\":\"$HOSTNAME\",\"content\":\"$IP\",\"ttl\":1,\"proxied\":false}"
  echo "DNS record for $HOSTNAME updated to IPv6: $IP"
else
  # 如果记录不存在，创建新的 DNS 记录
  echo "No existing DNS record found. Creating new record..."
  curl -s -X POST "https://api.cloudflare.com/client/v4/zones/$ZONE_ID/dns_records" \
       -H "Authorization: Bearer $CF_API_TOKEN" \
       -H "Content-Type: application/json" \
       --data "{\"type\":\"AAAA\",\"name\":\"$HOSTNAME\",\"content\":\"$IP\",\"ttl\":1,\"proxied\":false}"
  echo "DNS record for $HOSTNAME created with IPv6: $IP"
fi

echo "$(date) - Script executed" >> /data/data/com.termux/files/home/ddns.log

echo ""
echo ""
