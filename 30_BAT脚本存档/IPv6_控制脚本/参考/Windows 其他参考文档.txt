		DefaultHopLimit               : 128

		AddressMaskReply              : Disabled
		UseTemporaryAddresses         : Enabled
		MaxTemporaryDadAttempts       : 3
		MaxTemporaryValidLifetime     : 7.00:00:00 默认7天
		MaxTemporaryPreferredLifetime : 1.00:00:00 默认1天 #最低11分钟 不能比MaxTemporaryDesyncTime小
		TemporaryRegenerateTime       : 00:00:05
		MaxTemporaryDesyncTime        : 00:10:00
		DeadGatewayDetection          : Enabled

	Set-NetIPv6Protocol -DeadGatewayDetection Enabled
