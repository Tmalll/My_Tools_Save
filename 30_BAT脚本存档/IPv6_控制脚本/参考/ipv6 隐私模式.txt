echo ipv6 是否使用 隐私模式 临时地址
netsh interface ipv6 set privacy state=enable
netsh interface ipv6 set privacy state=disable

echo ipv6 是否使用 随机标识符
随机标识符只能在全局范围内停用。命令的输出：
netsh interface ipv6 set global randomizeidentifiers=disable
netsh interface ipv6 set global randomizeidentifiers=enable


echo 设置临时地址有效期
netsh interface ipv6 set privacy maxvalidlifetime=1d
